
#include <LiquidCrystal.h>

// LCD CONNECTIONS
LiquidCrystal lcd(7, 8, 5, 4, 3, 2);

// POTENTIOMETERS
int pot1 = A0;
int pot2 = A1;
int pot3 = A2;

// LED AND BUZZER PINS
const int buzzer = 13;
const int redLED = 12;
const int yellowLED = 11;
const int greenLED = 10;

void setup() {
  Serial.begin(9600);

  lcd.begin(16, 2);
  lcd.clear();

  pinMode(buzzer, OUTPUT);
  pinMode(redLED, OUTPUT);
  pinMode(yellowLED, OUTPUT);
  pinMode(greenLED, OUTPUT);
}

void loop() {

  // READ POTENTIOMETERS
  int raw1 = analogRead(pot1);
  int raw2 = analogRead(pot2);
  int raw3 = analogRead(pot3);

  // CONVERT SENSOR VALUES
  float temperature = raw1 * 150.0 / 1023.0;
  float pressure = raw2 * 10.0 / 1023.0;
  float vibration = raw3 * 12.0 / 1023.0;

  String status = "NORMAL";

  // CHECK MACHINE CONDITION
  if (temperature >= 90 || pressure >= 9 || vibration >= 8) {
    status = "CRITICAL";
  }
  else if (temperature >= 80 || pressure >= 7 || vibration >= 5) {
    status = "WARNING";
  }

  // RESET ALL OUTPUTS
  digitalWrite(redLED, LOW);
  digitalWrite(yellowLED, LOW);
  digitalWrite(greenLED, LOW);
  noTone(buzzer);

  // ALERT SYSTEM
  if (status == "NORMAL") {
    digitalWrite(greenLED, HIGH);
  }
  else if (status == "WARNING") {
    digitalWrite(yellowLED, HIGH);
  }
  else if (status == "CRITICAL") {
    digitalWrite(redLED, HIGH);
    tone(buzzer, 1000);
  }

  // LCD FIRST ROW
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(temperature, 1);
  lcd.print(" P:");
  lcd.print(pressure, 1);
  lcd.print("  ");

  // LCD SECOND ROW
  lcd.setCursor(0, 1);
  lcd.print("V:");
  lcd.print(vibration, 1);
  lcd.print(" ");
  lcd.print(status);
  lcd.print("  ");

  // SERIAL MONITOR
  Serial.print("Temperature: ");
  Serial.print(temperature, 1);
  Serial.print(" C | Pressure: ");
  Serial.print(pressure, 1);
  Serial.print(" bar | Vibration: ");
  Serial.print(vibration, 1);
  Serial.print(" mm/s | Status: ");
  Serial.println(status);

  delay(500);
}
